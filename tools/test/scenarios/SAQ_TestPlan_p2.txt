[case:r120_gfx_poc]
desc = 第 120 轮（P2）：原版 SWF 完整 PoC —— 7 → 8 个 tab + 拦截 + 列表数据注入（无 SWF 覆盖形态）
;  为什么 menu.open 之后用 wait：原版环境没有「推送成功」行可等桥就绪（我们 SWF
;  的推送在 P2 下必然失败）—— 用固定 1800 ms 留出 Movie/ASMovieRoot 解析时间
;  （P1.5 实测：菜单打开后桥在 ~1 秒内就绪）。桥没通 ⇒ 探针行会写「桥没通（…）」
;  ⇒ 首条断言 FAIL，能一眼看出是时序而不是能力问题。
step = ping
step = menu.open
step = wait 1800
step = ui.research3
step = assert.log 界面研究探针3 .*Menu_mc=ok scope=prev timeout=8000
step = assert.log 界面研究探针3 .*扩tab=ok.*切3=ok.*切7=ok.*切0=ok.*清理=ok.*注入=ok scope=case timeout=8000
;  给「眼睛」留时间（★ 第 123 轮起 8 → 30 秒）：菜单开着时的界面状态 =
;  8 个 tab（前 7 个 `$SAQ-keep0..6` 假数据 + 第 8 个 `SAQ-PoC`）
;  + 列表 3 条 `SAQ-PoC-Item 0/1/2`；menu.close 后随 Movie 销毁。
step = wait 30000
step = menu.close
;  ★★★ 第 127 轮（提示时机）：关菜单后的静默窗口 —— 「UI 通道不可用」的 HUD 提示
;  只等菜单关闭后再发（首发 + 补发 2 次 ≈ 6 秒）。这 10 秒留给「眼睛」：关掉菜单后
;  HUD 上应弹出中英提示「可接任务界面未生效：… / Available Quests UI not active …」。
;  放在这条用例末尾（而不是 r125 之后）：**第一次关菜单**就会触发提示，
;  这段静默期让 3 次提示完整播完、不被下一条用例的 menu.open 打断。
step = wait 10000

[case:r125_ui_conflict]
desc = 第 125 轮（路线 D · 冲突检测）：原版 SWF 下判定「UI 通道不可用」+ 停推降噪（不再空转 14 次重试）
;  判据：① `UI 通道不可用（第 N 次推送失败后判定）：…` 一行（产品 WARN —— 容它落在
;  上一步附近，scope=case 覆盖整条用例）；② **不再出现**「推送放弃：重试 14 次」——
;  判定发生在第 2 次正式失败后立即停推（旧行为会一路退避重试到预算耗尽）。
;  注：判定的两个前提 = 桥解析成功 + `_root.SAQ_Report` 调用失败 —— P2 环境（原版 SWF）正是该形态。
;  提示 GLOB（SAQ_UiNotice）不断言：进程内只请求一次，r120 用例窗口里已请求过。
step = ping
step = menu.open
step = wait 3500
step = assert.log UI 通道不可用 scope=case timeout=8000
step = assert.nolog 推送放弃 scope=case timeout=1000
step = menu.close

[case:r130_gfx_migrate]
desc = 第 130 轮（功能迁移探针 v4）：U5 按键接管 / U6 引擎条目直读 / U7 就地刷新 / U8 语言+渲染文本 / U9 关闭原语 / U10 类通道
;  判据（`docs/15` 十一节的 U5~U10）：
;    4a = 对象/数据层：读条目（GetDataForEntry / selectedEntry）/ 语言（引擎任务名 CJK）/
;         类通道（applicationDomain → BSUIDataManager + hasEventListener 静态调用 +
;         Subscribe("QuestData")）/ 造对象（CreateObject 带类名 → UserEventData →
;         UserEventManager → ButtonBaseData 三级真 AS3 实例）/ 接管（SetButtonData 换到
;         REJECT 按钮 → HandleUserEvent("R3") 程序化触发 → 我们的回调收到）/
;         注入（2 条真实字段集条目）/ 选中（selectedIndex → selectedEntry 读回）/
;         置灰（bCanShowOnMap=false ⇒ SET COURSE Enabled=0；=true ⇒ 1）。
;    4b = 渲染层（与 4a 之间隔一帧）：就地刷新（竖条 Inactive→Active）/ 渲染文本
;         （clip 的 TextField）/ 关菜单入口（ProcessUserEvent 返回 false 且菜单仍在）。
;  判据里 `类通道`/`造对象`/`接管` 用 (ok|fail) 宽匹配：这三项是**能力边界测量**
;  （任一 fail 都有对应的 fallback，见 docs/15 11.4-⑥），必须「有结论」而不是「猜 ok」。
;  眼睛：列表被替换成 2 条 `SAQ-Mig-0/1`（当前 tab 掩码 = 1<<6）；选中一条 ⇒ 右侧描述
;    面板显示的就是 `sDescription` 字段（Probe 用的 ASCII 文案，避免英文环境缺中文字形）；
;    破坏性副作用随 `menu.close` 清理（第 27/50 轮定案；重开后 = 原版数据）。
;  ★ 探针副作用（仅本次会话）：REJECT 按钮的 Data 被换成探针造的实例（该按钮平时
;    不可见；「换 Data 是否被接受」本身就是 U5 的判据 —— 菜单关闭随 Movie 销毁还原）。
step = ping
step = menu.open
step = wait 1800
step = ui.research4
step = assert.log 界面研究探针4a .*Menu_mc=ok scope=prev timeout=10000
step = assert.log 界面研究探针4a .*读条目=ok.*语言=.*类通道=(ok|fail).*造对象=(ok|fail).*接管=(ok|fail|未试).*注入=ok.*选中=ok.*置灰=ok scope=case timeout=10000
;  U7（就地刷新）是渲染层证据：clip 的 itemIndex 由 BSScrollingContainer.Update 在
;  帧推进时写 ⇒ 注入与读回之间必须隔一帧（1200 ms 留够菜单帧推进）。
step = wait 1200
step = ui.research4b
step = assert.log 界面研究探针4b .*Menu_mc=ok scope=prev timeout=10000
step = assert.log 界面研究探针4b .*刷新=ok.*文本=ok.*关菜单入口=ok scope=case timeout=10000
;  眼睛窗口（30 秒）：2 条 SAQ-Mig 条目 + 描述面板（sDescription 字段）。
step = wait 30000
step = menu.close
